import { InspectorModal } from '@/components/vault/InspectorModal';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
} from '@/components/ui/alert-dialog';
import { ShieldCheck, Loader2, Terminal, Lock, KeyRound, Sparkles, Home } from 'lucide-react';

export const VaultPage: React.FC = () => {
  const { lockState, resetEntireVault, lockVault } = useVault();
  const [isInspectorOpen, setIsInspectorOpen] = useState(false);
  const [isResetConfirmOpen, setIsResetConfirmOpen] = useState(false);
  // Show landing page by default if user is unauthenticated, but can switch directly to setup/login
  const [showLanding, setShowLanding] = useState(true);

  // If vault is already unlocked, show dashboard directly
  const isUnlocked = lockState === 'unlocked';

  return (
    <div className="min-h-screen bg-background text-foreground flex flex-col justify-between selection:bg-cyan-500/30 selection:text-white">
      {/* SaaS Top Navigation Bar */}
      <header className="border-b border-border/70 bg-card/60 backdrop-blur-md sticky top-0 z-30">
        <div className="max-w-7xl mx-auto px-4 h-14 flex items-center justify-between text-xs">
          {/* Logo & Brand */}
          <div className="flex items-center gap-3">
            <button
              type="button"
              onClick={() => setShowLanding(true)}
              className="flex items-center gap-2.5 text-foreground hover:opacity-90 transition-opacity"
            >
              <div className="w-8 h-8 rounded-xl bg-gradient-to-br from-cyan-500 to-purple-600 flex items-center justify-center text-white shadow-md shadow-cyan-500/20">
                <ShieldCheck className="h-4 w-4" />
              </div>
              <span className="font-extrabold text-sm tracking-tight gradient-cyber-text">
                SecureVault
              </span>
            </button>
            <Badge variant="outline" className="hidden sm:inline-flex text-[10px] font-mono border-primary/30 text-primary bg-primary/10">
              AES-256-GCM Zero-Knowledge
            </Badge>
          </div>

          {/* Right Navigation Controls */}
          <div className="flex items-center gap-2.5">
            {!isUnlocked && (
              <button
                type="button"
                onClick={() => setShowLanding(!showLanding)}
                className="hidden sm:flex items-center gap-1.5 text-xs text-muted-foreground hover:text-foreground transition-colors px-2 py-1"
              >
                {showLanding ? (
                  <>
                    <KeyRound className="h-3.5 w-3.5 text-cyan-400" />
                    <span>{lockState === 'uninitialized' ? 'Register' : 'Login'}</span>
                  </>
                ) : (
                  <>
                    <Home className="h-3.5 w-3.5 text-purple-400" />
                    <span>Home</span>
                  </>
                )}
              </button>
            )}

            <Button
              variant="outline"
              size="sm"
              onClick={() => setIsInspectorOpen(true)}
              className="h-8 rounded-xl text-xs border-border/80 hover:border-cyan-500/50 text-foreground"
            >
              <Terminal className="h-3.5 w-3.5 mr-1 text-cyan-400" />
              <span className="hidden sm:inline">Inspector</span>
            </Button>

            {isUnlocked ? (
              <Button
                size="sm"
                variant="outline"
                onClick={() => lockVault('Vault locked by user.')}
                className="h-8 rounded-xl text-xs border-border hover:border-destructive/50 text-destructive hover:bg-destructive/10"
              >
                <Lock className="h-3.5 w-3.5 mr-1" /> Lock
              </Button>
            ) : (
              <Button
                size="sm"
                onClick={() => setShowLanding(false)}
                className="cyber-btn-primary h-8 px-3 rounded-xl text-xs font-medium"
              >
                {lockState === 'uninitialized' ? 'Get Started' : 'Unlock Vault'}
              </Button>
            )}
          </div>
        </div>
      </header>

      {/* Main Content Area based on Lock State & Active View */}
      <main className="flex-1 flex flex-col justify-center">
        {lockState === 'loading' && (
          <div className="flex flex-col items-center justify-center p-12 space-y-3">
            <Loader2 className="h-7 w-7 animate-spin text-cyan-400" />
            <p className="text-xs text-muted-foreground font-mono">Initializing local Web Crypto engine &amp; IndexedDB...</p>
          </div>
        )}

        {lockState !== 'loading' && isUnlocked && (
          <VaultDashboard onOpenInspector={() => setIsInspectorOpen(true)} />
        )}

        {lockState !== 'loading' && !isUnlocked && showLanding && (
          <LandingPage
            isInitialized={lockState === 'locked'}
            onGetStarted={() => setShowLanding(false)}
            onOpenInspector={() => setIsInspectorOpen(true)}
          />
        )}

        {lockState !== 'loading' && !isUnlocked && !showLanding && lockState === 'uninitialized' && (
          <MasterPasswordSetup
            onOpenInspector={() => setIsInspectorOpen(true)}
            onBackToLanding={() => setShowLanding(true)}
          />
        )}

        {lockState !== 'loading' && !isUnlocked && !showLanding && lockState === 'locked' && (
          <MasterPasswordUnlock
            onOpenInspector={() => setIsInspectorOpen(true)}
            onResetRequested={() => setIsResetConfirmOpen(true)}
            onBackToLanding={() => setShowLanding(true)}
          />
        )}
      </main>

      {/* Modern SaaS Footer */}
      <footer className="border-t border-border/70 py-5 text-center text-xs text-muted-foreground bg-card/40 backdrop-blur-sm">
        <div className="max-w-7xl mx-auto px-4 flex flex-col sm:flex-row items-center justify-between gap-3">
          <div className="flex items-center gap-2">
            <span className="font-semibold text-foreground">SecureVault</span>
            <span>&bull;</span>
            <span>Zero-Knowledge Browser Security</span>
          </div>

          <p className="font-mono text-[11px] text-muted-foreground">
            Web Crypto API (PBKDF2 100k + AES-256-GCM) &bull; IndexedDB Local
          </p>
        </div>
      </footer>

      {/* Reset Vault Confirmation Dialog */}
      <AlertDialog open={isResetConfirmOpen} onOpenChange={setIsResetConfirmOpen}>
        <AlertDialogContent className="max-w-[calc(100%-2rem)] md:max-w-md cyber-card rounded-2xl border-destructive/40">
          <AlertDialogHeader>
            <AlertDialogTitle className="text-destructive">Reset Database &amp; Wipe Vault?</AlertDialogTitle>
            <AlertDialogDescription className="text-xs text-muted-foreground text-pretty">
              This will completely delete the IndexedDB store (<code>SecureVaultDB</code>). You will be able to set a new master password and start from scratch.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel className="rounded-xl text-xs">Cancel</AlertDialogCancel>
            <AlertDialogAction
              onClick={async () => {
                setIsResetConfirmOpen(false);
                await resetEntireVault();
              }}
              className="bg-destructive text-destructive-foreground hover:bg-destructive/90 rounded-xl text-xs"
            >
              Confirm Wipe
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>

      {/* Technical Inspector Modal */}
      <InspectorModal
        open={isInspectorOpen}
        onOpenChange={setIsInspectorOpen}
      />
    </div>
  );
};

export default VaultPage;

# Theme Name: Cathedral Gothic

# Vibe & Description:
Medieval cathedral aesthetics, with deep purple, blood red, and black setting the tone, paired with gold lines, pointed arches, and rose windows for a dark, solemn, mysterious classical mood.

# Color

- Use deep purple, blood red, black, and gold exclusively; avoid pure white, light gray, pink, green, blue, and neon colors.
- Make black (#000000) or deep purple (#2D1B4E) the main background, with blood red (#8B0000) used in spots to create a heavy, religious mural-like feel. Use gold (#FFD700 / #c9a227) for text and decoration, and semi-transparent gold for borders (border-[#c9a227]/40), presenting the solemn visual of a "dark cathedral".

# Font

- Use “SourceHanSerifCN-Bold” as the heading font. (https://resource-static.bj.bcebos.com/fonts-skill/SourceHanSerifCN-Bold.woff2)
- Use “AlibabaPuHuiTi-Regular” as the body font. (https://resource-static.bj.bcebos.com/fonts-skill/AlibabaPuHuiTi_Regular.ttf)

# Animation

- On touch, do not use a lively bounce; use a gold halo and deeper shadow as feedback, keeping it solemn and restrained. On tap, allow a slight scale to `active:scale-95`.
- When elements appear, use a slow fade-in or dark-area emergence effect, like candlelight illuminating carved stone patterns. Use `transition-all duration-300 ease-in-out` uniformly for transitions.

# Layout

- The page should be vertically oriented, symmetrical, and ritualistic, simulating a cathedral central axis. On mobile, keep a single-column layout, leaving ample breathing room between cards and sections.
- Section edges may incorporate pointed arches, thin gold lines, or manuscript-style borders. Use sharp-cornered cards and avoid rounded contours; establish hierarchy through dark-light variation, gold outlines, and font-size differences.

# Elements

- Buttons use transparent backgrounds, gold outlines, and serif text; on hover or press, a semi-transparent gold background and dark text appear.
- Cards use `bg-[#0a0a0a]/90`, `border-2 border-[#c9a227]/40`, `rounded-none`, with shadows rendered as deep dark diffusion or gold halos.
- Prefer pointed arches, rose windows, flying buttress lines, and manuscript borders for decorative elements to strengthen the medieval Gothic atmosphere.
- Divider lines, icons, and navigation should emphasize thin gold lines, avoiding modern rounded corners, soft cartoon forms, and highly saturated bright colors.
