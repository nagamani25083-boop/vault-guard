              Delete

      <EntryFormDialog

        entryToEdit={editingEntry}
        defaultCategory={

            await addEntry(data);
          }
        }}
      />

      {/* Standalone Password Generator */}
      <PasswordGeneratorDialog
        open={isGeneratorOpen}
        onOpenChange={setIsGeneratorOpen}
      />
    </div>
  );
};
